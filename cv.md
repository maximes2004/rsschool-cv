### Code Example ###

function isPrime(num) {
    // Числа меньше или равные 1 не являются простыми.
    if (num <= 1) {
        return false;
    }

    // Проверяем делимость числа на все целые числаот 2 до кв. корня из числа.
    for (let i = 2; i <= Math.sqrt(num); i++) {
        if (num % i === 0) {
            // Если найден делитель, число не простое.
            return false;
        }
    }

    // Если цикл завершился без нахождения делителей, число простое.
    return true;
}


// Проверка

console.log(`Число 19 является простым: ${isPrime(19)}`); // Ожидается true
