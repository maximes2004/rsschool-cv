function isPrime(num) {
    // Numbers less than or equal to 1 are not prime.
    if (num <= 1) {
        return false;
    }

    // We check the divisibility of a number by all integers from 2 to the square root of the number.
    for (let i = 2; i <= Math.sqrt(num); i++) {
        if (num % i === 0) {
            // If a divisor is found, the number is not prime.
            return false;
        }
    }

    // If the cycle terminates without finding any divisors, the number is prime.
    return true;
}


// Testing

console.log(`The number 19 is prime: ${isPrime(19)}`); // Expected true
